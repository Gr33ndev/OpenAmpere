#include <tunables/global>

# AppArmor profile of the OpenAmpere app on Home Assistant OS (#300). The Supervisor puts the slug of the app in place
# of the name below. Files as in the template of the Home Assistant docs (the container's own files, /data and the
# config folder of Home Assistant for the integration); the rest is limited to what OpenAmpere uses.
profile openampere flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  file,

  # only for ha-app-start, which runs as root: hand /data to the app user, install the integration, switch to the
  # app user without any capabilities (setpriv)
  capability chown,
  capability dac_override,
  capability dac_read_search,
  capability fowner,
  capability setgid,
  capability setuid,
  capability setpcap,

  # web app and API (8080, 8443), Modbus TCP to the inverter, Supervisor API, DNS, own addresses (getaddrinfo)
  network inet stream,
  network inet6 stream,
  network inet dgram,
  network inet6 dgram,
  network netlink raw,
  network unix,

  # start, stop and restart by the Supervisor, signals between the processes of the container
  signal,

  # everything that runs as the app user starts through setpriv: OpenAmpere, the backup commands and the health
  # check. From there on the profile "server" applies, without the config folder of Home Assistant: only the start
  # step as root needs it, to install the integration.
  /usr/bin/setpriv cx -> server,

  profile server flags=(attach_disconnected,mediate_deleted) {
    #include <abstractions/base>

    file,
    deny /homeassistant/ rw,
    deny /homeassistant/** rwlkmx,

    # setpriv switches to the app user and gives up every capability; OpenAmpere itself has none
    capability setgid,
    capability setuid,
    capability setpcap,

    network inet stream,
    network inet6 stream,
    network inet dgram,
    network inet6 dgram,
    network netlink raw,
    network unix,

    signal,
  }
}
